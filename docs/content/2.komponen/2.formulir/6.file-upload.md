## Penggunaan di Berbagai Teknologi

Salin contoh sesuai teknologi proyek Anda. Semua versi memakai class dari `@majalengka/css`. Komponen siap pakai tersedia di `@majalengka/vue`; untuk React, Angular, Laravel, dan PHP, contoh di bawah sudah berupa pembungkus kecil yang bisa langsung disalin ke proyek.

::code-group

```html [HTML]
<div class="mjl-form-field">
  <label class="mjl-label" for="ktp">Foto KTP Elektronik</label>
  <div class="mjl-file-upload">
    <p><strong>Klik untuk memilih berkas</strong> atau seret ke sini</p>
    <p class="mjl-help-text">JPG, PNG, atau PDF, maksimal 2 MB</p>
    <input type="file" id="ktp" name="ktp" accept="image/*,.pdf">
  </div>
</div>
```

```vue [Vue]
<script setup lang="ts">
import { ref } from 'vue';
import { MjlFileUpload } from '@majalengka/vue';

const berkas = ref<FileList | null>(null);
</script>

<template>
  <label class="mjl-label" for="ktp">Foto KTP Elektronik</label>
  <MjlFileUpload id="ktp" accept="image/*,.pdf" hint="JPG, PNG, atau PDF, maksimal 2 MB" @update:model-value="berkas = $event" />
</template>
```

```tsx [React]
import { useState } from 'react';

export function UnggahKtp() {
  const [nama, setNama] = useState('');
  return (
    <div className="mjl-form-field">
      <label className="mjl-label" htmlFor="ktp">Foto KTP Elektronik</label>
      <div className="mjl-file-upload">
        <p><strong>{nama || 'Klik untuk memilih berkas'}</strong></p>
        <p className="mjl-help-text">JPG, PNG, atau PDF, maksimal 2 MB</p>
        <input type="file" id="ktp" accept="image/*,.pdf" onChange={(e) => setNama(e.target.files?.[0]?.name ?? '')} />
      </div>
    </div>
  );
}
```

```ts [Angular]
import { Component, signal } from '@angular/core';

@Component({
  selector: 'app-unggah-ktp',
  standalone: true,
  template: `
    <div class="mjl-form-field">
      <label class="mjl-label" for="ktp">Foto KTP Elektronik</label>
      <div class="mjl-file-upload">
        <p><strong>{{ fileName() || 'Klik untuk memilih berkas' }}</strong></p>
        <p class="mjl-help-text">JPG, PNG, atau PDF, maksimal 2 MB</p>
        <input type="file" id="ktp" accept="image/*,.pdf" (change)="onFile($event)" />
      </div>
    </div>
  `,
})
export class UnggahKtp {
  fileName = signal('');
  onFile(e: Event) { this.fileName.set((e.target as HTMLInputElement).files?.[0]?.name ?? ''); }
}
```
