## Penggunaan di Berbagai Teknologi

Salin contoh sesuai teknologi proyek Anda. Semua versi memakai class dari `@majalengka/css`. Komponen siap pakai tersedia di `@majalengka/vue`; untuk React, Angular, Laravel, dan PHP, contoh di bawah sudah berupa pembungkus kecil yang bisa langsung disalin ke proyek.

```tsx [React]
import type { ReactNode } from 'react';

export function MjlCard({ title, desc, href, footer }: { title: string; desc?: string; href?: string; footer?: ReactNode }) {
  const isi = (
    <>
      <div className="mjl-card__header">
        <h3 className="mjl-card__title">{title}</h3>
        {desc && <p className="mjl-card__desc">{desc}</p>}
      </div>
      {footer && <div className="mjl-card__footer">{footer}</div>}
    </>
  );
  return href
    ? <a href={href} className="mjl-card mjl-card--interactive">{isi}</a>
    : <div className="mjl-card">{isi}</div>;
}
```

```ts [Angular]
import { Component, input } from '@angular/core';
import { RouterLink } from '@angular/router';

@Component({
  selector: 'mjl-card',
  standalone: true,
  imports: [RouterLink],
  template: `
    <a [routerLink]="link()" class="mjl-card mjl-card--interactive">
      <div class="mjl-card__header">
        <h3 class="mjl-card__title">{{ title() }}</h3>
        @if (desc()) { <p class="mjl-card__desc">{{ desc() }}</p> }
      </div>
      <div class="mjl-card__footer"><ng-content /></div>
    </a>
  `,
})
export class MjlCard {
  title = input.required<string>();
  desc = input<string>();
  link = input.required<string>();
}
```

```blade [Laravel]

      </div>
    </a>
  @endforeach
</div>
```

  </a>
<?php endforeach; ?>
