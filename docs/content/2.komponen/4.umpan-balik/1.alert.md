## Penggunaan di Berbagai Teknologi

Salin contoh sesuai teknologi proyek Anda. Semua versi memakai class dari `@majalengka/css`. Komponen siap pakai tersedia di `@majalengka/vue`; untuk React, Angular, Laravel, dan PHP, contoh di bawah sudah berupa pembungkus kecil yang bisa langsung disalin ke proyek.

```vue [Vue]

import { MjlAlert } from '@majalengka/vue';

const tampil = ref(true);
</script>

<template>
  <MjlAlert v-if="tampil" variant="success" title="Berkas berhasil terkirim" dismissible @dismiss="tampil = false">

  </MjlAlert>
```

```tsx [React]
import type { ReactNode } from 'react';

type Variant = 'info' | 'success' | 'warning' | 'danger';

export function MjlAlert({ variant = 'info', title, children }: { variant?: Variant; title?: string; children: ReactNode }) {
  // Pesan error/peringatan diumumkan segera (alert); info/sukses cukup sopan (status).
  const role = variant === 'danger' || variant === 'warning' ? 'alert' : 'status';
  return (
    <div className={`mjl-alert mjl-alert--${variant}`} role={role}>
      <div className="mjl-alert__content">
        {title && <p className="mjl-alert__title">{title}</p>}
        <div>{children}</div>
      </div>
    </div>
  );
}
```

```ts [Angular]
import { Component, computed, input } from '@angular/core';

@Component({
  selector: 'mjl-alert',
  standalone: true,
  template: `
    <div class="mjl-alert mjl-alert--{{ variant() }}" [attr.role]="role()">
      <div class="mjl-alert__content">
        @if (title()) { <p class="mjl-alert__title">{{ title() }}</p> }
        <ng-content />
      </div>
    </div>
  `,
})
export class MjlAlert {
  variant = input<'info' | 'success' | 'warning' | 'danger'>('info');
  title = input<string>();
  role = computed(() => (this.variant() === 'danger' || this.variant() === 'warning' ? 'alert' : 'status'));
}
```

```blade [Laravel]
{{-- Tampilkan flash message dari controller: return back()->with('sukses', '...') --}}
@if (session('sukses'))
  <div class="mjl-alert mjl-alert--success" role="status">
    <div class="mjl-alert__content">
      <p class="mjl-alert__title">Berhasil</p>
      <p>{{ session('sukses') }}</p>
    </div>
  </div>
@endif

@if ($errors->any())
  <div class="mjl-alert mjl-alert--danger" role="alert">
    <div class="mjl-alert__content">
      <p class="mjl-alert__title">Periksa kembali isian Anda</p>
      <ul>@foreach ($errors->all() as $error)<li>{{ $error }}</li>@endforeach</ul>
    </div>
  </div>
@endif
```

```php [PHP]
<?php session_start(); ?>
<?php if (!empty($_SESSION['pesan_sukses'])): ?>
  <div class="mjl-alert mjl-alert--success" role="status">
    <div class="mjl-alert__content">
      <p class="mjl-alert__title">Berhasil</p>
      <p><?= htmlspecialchars($_SESSION['pesan_sukses'], ENT_QUOTES, 'UTF-8') ?></p>
    </div>
  </div>
  <?php unset($_SESSION['pesan_sukses']); ?>
<?php endif; ?>
```

## 2. Aturan Aksesibilitas ARIA

- **Pesan Bahaya / Error**: Gunakan `role="alert"` dan `aria-live="assertive"` agar pembaca layar segera menginterupsi dan membacakan kendala kepada pengguna tuna netra.
- **Pesan Sukses / Info**: Gunakan `role="status"` dan `aria-live="polite"` agar pembaca layar menyelesaikan kalimat aktif terlebih dahulu sebelum membacakan status baru.
