```tsx [React]
import { useEffect, useId, useRef, type ReactNode } from 'react';

export function MjlDrawer({ open, title, side = 'right', onClose, children }: { open: boolean; title: string; side?: 'left' | 'right' | 'bottom'; onClose: () => void; children: ReactNode }) {
  const ref = useRef<HTMLDialogElement>(null);
  const titleId = useId();

  useEffect(() => {
    const d = ref.current;
    if (open && !d?.open) d?.showModal();
    if (!open && d?.open) d.close();
  }, [open]);

  return (
    <dialog ref={ref} className="mjl-drawer-backdrop" aria-labelledby={titleId} onClose={onClose}
      onClick={(e) => e.target === e.currentTarget && onClose()}>
      <div className={`mjl-drawer mjl-drawer--${side}`}>
        <div className="mjl-drawer__header">
          <h2 id={titleId} className="mjl-drawer__title">{title}</h2>
          <button type="button" className="mjl-icon-btn" aria-label="Tutup" onClick={onClose}>✕</button>
        </div>
        <div className="mjl-drawer__body">{children}</div>
      </div>
    </dialog>
  );
}
```

```ts [Angular]
import { Component, ElementRef, effect, input, model, viewChild } from '@angular/core';

@Component({
  selector: 'mjl-drawer',
  standalone: true,
  template: `
    <dialog #dialog class="mjl-drawer-backdrop" aria-labelledby="drawer-title" (close)="open.set(false)"
      (click)="$event.target === dialog && open.set(false)">
      <div class="mjl-drawer mjl-drawer--{{ side() }}">
        <div class="mjl-drawer__header">
          <h2 id="drawer-title" class="mjl-drawer__title">{{ title() }}</h2>
          <button type="button" class="mjl-icon-btn" aria-label="Tutup" (click)="open.set(false)">✕</button>
        </div>
        <div class="mjl-drawer__body"><ng-content /></div>
      </div>
    </dialog>
  `,
})
export class MjlDrawer {
  open = model(false);
  title = input.required<string>();
  side = input<'left' | 'right' | 'bottom'>('right');
  dialog = viewChild.required<ElementRef<HTMLDialogElement>>('dialog');

  constructor() {
    effect(() => {
      const d = this.dialog().nativeElement;
      if (this.open() && !d.open) d.showModal();
      if (!this.open() && d.open) d.close();
    });
  }
}
```

::

Dibangun di atas `<dialog>` + `showModal()`: fokus terkunci di dalam panel, Esc menutup, konten di belakang tidak bisa diakses, dan fokus kembali ke tombol pemicu saat ditutup.
