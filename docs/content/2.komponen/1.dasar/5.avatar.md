## Penggunaan di Berbagai Teknologi

Salin contoh sesuai teknologi proyek Anda. Semua versi memakai class dari `@majalengka/css`. Komponen siap pakai tersedia di `@majalengka/vue`; untuk React, Angular, Laravel, dan PHP, contoh di bawah sudah berupa pembungkus kecil yang bisa langsung disalin ke proyek.

```tsx [React]
import { useState } from 'react';

const inisial = (nama: string) => nama.trim().split(/\s+/).slice(0, 2).map((k) => k[0]).join('');

export function MjlAvatar({ name, src, size = 'md' }: { name: string; src?: string; size?: 'xs' | 'sm' | 'md' | 'lg' | 'xl' }) {
  const [gagal, setGagal] = useState(false);
  return (
    <span className={`mjl-avatar mjl-avatar--${size} mjl-avatar--circle`} role="img" aria-label={name}>
      {src && !gagal
        ? <img className="mjl-avatar__image" src={src} alt="" onError={() => setGagal(true)} />
        : <span className="mjl-avatar__fallback" aria-hidden="true">{inisial(name)}</span>}
    </span>
  );
}

// Pemakaian: <MjlAvatar name="Siti Nurhaliza" src="/foto/siti.jpg" size="lg" />
```

```ts [Angular]
import { Component, computed, input, signal } from '@angular/core';

@Component({
  selector: 'mjl-avatar',
  standalone: true,
  template: `
    <span class="mjl-avatar mjl-avatar--{{ size() }} mjl-avatar--circle" role="img" [attr.aria-label]="name()">
      @if (src() && !failed()) {
        <img class="mjl-avatar__image" [src]="src()" alt="" (error)="failed.set(true)" />
      } @else {
        <span class="mjl-avatar__fallback" aria-hidden="true">{{ initials() }}</span>
      }
    </span>
  `,
})
export class MjlAvatar {
  name = input.required<string>();
  src = input<string>();
  size = input<'xs' | 'sm' | 'md' | 'lg' | 'xl'>('md');
  failed = signal(false);
  initials = computed(() => this.name().trim().split(/\s+/).slice(0, 2).map((k) => k[0]).join(''));
}
```

```blade [Laravel]
{{-- resources/views/components/mjl/avatar.blade.php --}}
@props(['name', 'src' => null, 'size' => 'md'])
<span {{ $attributes->class("mjl-avatar mjl-avatar--{$size} mjl-avatar--circle") }} role="img" aria-label="{{ $name }}">
  @if ($src)
    <img class="mjl-avatar__image" src="{{ $src }}" alt="">
  @else
    <span class="mjl-avatar__fallback" aria-hidden="true">
      {{ collect(explode(' ', $name))->take(2)->map(fn ($k) => mb_substr($k, 0, 1))->implode('') }}
    </span>
  @endif
</span>

{{-- Pemakaian --}}
<x-mjl.avatar :name="$petugas->nama" :src="$petugas->foto_url" size="lg" />
```

```php [PHP]
<?php
$nama = 'Siti Nurhaliza';
$inisial = implode('', array_map(fn ($k) => mb_substr($k, 0, 1), array_slice(explode(' ', $nama), 0, 2)));
?>
<span class="mjl-avatar mjl-avatar--md mjl-avatar--circle" role="img" aria-label="<?= htmlspecialchars($nama, ENT_QUOTES, 'UTF-8') ?>">
  <span class="mjl-avatar__fallback" aria-hidden="true"><?= htmlspecialchars($inisial, ENT_QUOTES, 'UTF-8') ?></span>
</span>
```

::

`name` wajib: dipakai sebagai nama yang dibacakan pembaca layar dan sumber inisial cadangan.
